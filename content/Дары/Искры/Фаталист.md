// --- НАСТРОЙКИ ПУТЕЙ ---
const loreFolder = "Shattered Kingdoms/02 World/04 Magic and Powers/01 Sources of Power/02 Gifts and Sparks";
const artFolder = "Shattered Kingdoms/03 Rules and Tools/Sources/tokens/sparks"; // Папка с артами Искр

// --- ПОЛУЧЕНИЕ ДАННЫХ ---
const s = dv.current();
const sparkName = s.file.name;

// --- 1. ВЫВОД КАРТИНКИ В РАМКЕ (ЧЕРЕЗ BASE64) ---
const validExtensions = ["png", "jpg", "jpeg", "webp"];

const imageFile = app.vault.getFiles().find(f => 
    f.path.startsWith(artFolder) && 
    f.basename === sparkName && 
    validExtensions.includes(f.extension.toLowerCase())
);

if (imageFile) {
    try {
        const arrayBuffer = await app.vault.readBinary(imageFile);
        const base64String = btoa(
            new Uint8Array(arrayBuffer).reduce((data, byte) => data + String.fromCharCode(byte), '')
        );
        const ext = imageFile.extension.toLowerCase();
        let mimeType = 'image/jpeg'; 
        if (ext === 'png') mimeType = 'image/png';
        else if (ext === 'webp') mimeType = 'image/webp';
        
        const dataUrl = `data:${mimeType};base64,${base64String}`;
        
        const wrapper = dv.el("div", "");
        wrapper.style.textAlign = "center";
        wrapper.style.marginBottom = "20px";

        const frame = wrapper.createEl("div");
        frame.style.display = "inline-block";
        frame.style.padding = "4px";

        const img = frame.createEl("img");
        img.src = dataUrl;
        img.style.width = "180px";
        img.style.height = "auto";
        img.style.borderRadius = "8px";
        img.style.display = "block";
        img.style.objectFit = "cover";
        
    } catch (e) {
        console.error(`Ошибка загрузки картинки для ${sparkName}:`, e);
    }
}

// --- 2. ССЫЛКА НА ЛОР ---
const loreLink = dv.fileLink(`${loreFolder}/${sparkName}`);
dv.paragraph(`> Лор: ${loreLink}`);

// --- 3. ВЫВОД ПАРАМЕТРОВ ИСКРЫ ---
// Вместо таблицы физиологии выводим красивый блок с параметрами
const statsContainer = dv.el("div", "");
statsContainer.style.width = "100%";
statsContainer.style.maxWidth = "800px";
statsContainer.style.margin = "0 auto 20px auto";
statsContainer.style.padding = "15px";
statsContainer.style.backgroundColor = "var(--background-secondary)";
statsContainer.style.borderRadius = "8px";
statsContainer.style.border = "1px solid var(--background-modifier-border)";

const statsTable = statsContainer.createEl("table");
statsTable.style.width = "100%";
statsTable.style.borderCollapse = "collapse";
statsTable.style.textAlign = "left";

const addRow = (label, value) => {
    const tr = statsTable.createEl("tr");
    
    const td1 = tr.createEl("td", { text: label });
    td1.style.padding = "8px";
    td1.style.fontWeight = "bold";
    td1.style.width = "35%";
    td1.style.color = "var(--text-muted)";
    
    const td2 = tr.createEl("td", { text: value });
    td2.style.padding = "8px";
    td2.style.fontWeight = "bold";
};

addRow("Домен", s.domain || "Не указан");
addRow("Бонусы к физиологии", s.phys_bonus || "Нет");
addRow("Кость Здоровья (HP)", s.hp_die || "Не указана");
addRow("Кость Ментального Здоровья (SP)", s.sp_die || "Не указана");

// --- 4. ВЫВОД БЛАГОСЛОВЛЕНИЙ (СПИСКОМ) ---
if (s.blessings) {
    let blessText = "";
    if (Array.isArray(s.blessings)) {
        blessText = s.blessings.map(b => `> - ${b}`).join("\n");
    } else {
        blessText = `> ${s.blessings}`;
    }
    // Используем Callout с галочкой/подсказкой для хороших эффектов
    dv.paragraph(`> [!success] Благословения Искры\n${blessText}`);
}

// --- 5. ВЫВОД ОКОВ (СПИСКОМ) ---
if (s.chains) {
    let chainsText = "";
    if (Array.isArray(s.chains)) {
        chainsText = s.chains.map(o => `> - ${o}`).join("\n");
    } else {
        chainsText = `> ${s.chains}`;
    }
    // Используем Callout с черепом/предупреждением для негативных эффектов
    dv.paragraph(`> [!danger] Оковы Искры\n${chainsText}`);
}
